-- =====================================================================

-- ---------------------------------------------------------------------
-- 1. Tabela 'fleet' (Entregadores da Frota - Estrutura Antiga Garra)
-- ---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS public.fleet (
  id                  TEXT PRIMARY KEY,                  -- ID customizado ex: "#MB-1002"
  name                TEXT NOT NULL,
  vehicle             TEXT,
  plate               TEXT,
  status              TEXT NOT NULL DEFAULT 'Disponível',
  status_class        TEXT NOT NULL DEFAULT 'status-success',
  delivery            TEXT DEFAULT 'Nenhuma',            -- ID da tele ativa ou 'Nenhuma'
  battery             TEXT DEFAULT '100%',
  rating              NUMERIC(3, 2) DEFAULT 5.0,
  pin                 TEXT NOT NULL UNIQUE,              -- Código curto de login
  simultaneous_limit  INTEGER DEFAULT 1,
  lat                 DOUBLE PRECISION,
  lng                 DOUBLE PRECISION,
  last_seen           TIMESTAMP WITH TIME ZONE DEFAULT now()
);

-- ---------------------------------------------------------------------
-- 2. Tabela 'pending_deliveries' (Teles aguardando despacho - Legado Garra)
-- ---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS public.pending_deliveries (
  id          TEXT PRIMARY KEY,                          -- ID customizado ex: "#TELE-0001"
  client      TEXT DEFAULT 'Burger do Chef',
  dest_name   TEXT,
  address     TEXT,
  dist        TEXT,
  price       TEXT,
  payment     TEXT,
  cargo       TEXT,
  pickup_lat  DOUBLE PRECISION,
  pickup_lng  DOUBLE PRECISION,
  dest_lat    DOUBLE PRECISION,
  dest_lng    DOUBLE PRECISION,
  created_at  TIMESTAMP WITH TIME ZONE DEFAULT now()
);

-- ---------------------------------------------------------------------
-- 3. Tabela 'client_history' (Histórico Geral de Entregas - Legado Garra)
-- ---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS public.client_history (
  id           TEXT PRIMARY KEY,                         -- ID da tele
  dest_name    TEXT,
  address      TEXT,
  rider        TEXT,                                     -- Nome do motoboy
  dist         TEXT,
  price        TEXT,
  date         TEXT,                                     -- Data legível ("Hoje, 19:30" ou "21/06/2026")
  status       TEXT,
  status_class TEXT,
  payment_status TEXT NOT NULL DEFAULT 'Pendente',
  pickup_lat   DOUBLE PRECISION,
  pickup_lng   DOUBLE PRECISION,
  dest_lat     DOUBLE PRECISION,
  dest_lng     DOUBLE PRECISION,
  created_at   TIMESTAMP WITH TIME ZONE DEFAULT now()
);

-- ---------------------------------------------------------------------
-- 4. Tabela 'support_messages' (Chat de Suporte Central <-> Clientes)
-- ---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS public.support_messages (
  id            BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  client_email  TEXT NOT NULL,
  sender_role   TEXT NOT NULL CHECK (sender_role IN ('admin', 'client', 'rider')),
  sender_name   TEXT NOT NULL,
  message       TEXT NOT NULL,
  created_at    TIMESTAMP WITH TIME ZONE DEFAULT now()
);

-- ---------------------------------------------------------------------
-- Habilitar Row Level Security (RLS)
-- ---------------------------------------------------------------------
ALTER TABLE public.fleet ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.pending_deliveries ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.client_history ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.support_messages ENABLE ROW LEVEL SECURITY;

-- ---------------------------------------------------------------------
-- Criar políticas de acesso (Simples para MVP interno usando chave anon)
-- ---------------------------------------------------------------------
DROP POLICY IF EXISTS fleet_all ON public.fleet;
CREATE POLICY fleet_all ON public.fleet FOR ALL USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS pending_deliveries_all ON public.pending_deliveries;
CREATE POLICY pending_deliveries_all ON public.pending_deliveries FOR ALL USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS client_history_all ON public.client_history;
CREATE POLICY client_history_all ON public.client_history FOR ALL USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS support_messages_all ON public.support_messages;
CREATE POLICY support_messages_all ON public.support_messages FOR ALL USING (true) WITH CHECK (true);

-- ---------------------------------------------------------------------
-- Adicionar Tabelas ao Canal do Realtime (para atualização instantânea)
-- ---------------------------------------------------------------------
ALTER PUBLICATION supabase_realtime ADD TABLE public.fleet;
ALTER PUBLICATION supabase_realtime ADD TABLE public.pending_deliveries;
ALTER PUBLICATION supabase_realtime ADD TABLE public.client_history;
ALTER PUBLICATION supabase_realtime ADD TABLE public.support_messages;
